## 🔧 前端实现代码

### React版本

```jsx
// 文件：components/InterviewEngine.jsx

import React, { useState } from 'react';

const InterviewEngine({ stages, userAnswers, onAnswerChange }) => {
  const [currentStageIndex, setCurrentStageIndex] = useState(0);
  const [currentQuestionIndex, setCurrentQuestionIndex] = useState(0);

  const currentStage = stages[currentStageIndex];
  const currentQuestionId = currentStage.questions[currentQuestionIndex];
  const isLastQuestionInStage = currentQuestionIndex === currentStage.questions.length - 1;
  const isLastStage = currentStageIndex === stages.length - 1;

  // 获取当前题目到下一题的过渡信息
  const getTransition = () => {
    if (isLastQuestionInStage) return null;

    const fromQuestion = currentStage.questions[currentQuestionIndex];
    const toQuestion = currentStage.questions[currentQuestionIndex + 1];
    const transitionKey = `${fromQuestion}_to_${toQuestion}`;

    return currentStage.transitions[transitionKey] || null;
  };

  // 下一题按钮
  const handleNextQuestion = () => {
    const transition = getTransition();

    if (transition) {
      // 显示过渡脚本
      console.log(`过渡：${transition.script}`);
    }

    if (!isLastQuestionInStage) {
      // 阶段内：下一题
      setCurrentQuestionIndex(currentQuestionIndex + 1);
    } else {
      // 阶段间：下一阶段
      setCurrentStageIndex(currentStageIndex + 1);
      setCurrentQuestionIndex(0);
    }
  };

  // 上一题按钮
  const handlePreviousQuestion = () => {
    if (currentQuestionIndex > 0) {
      // 阶段内：上一题
      setCurrentQuestionIndex(currentQuestionIndex - 1);
    } else if (currentStageIndex > 0) {
      // 阶段间：上一阶段最后一题
      setCurrentStageIndex(currentStageIndex - 1);
      const prevStage = stages[currentStageIndex - 1];
      setCurrentQuestionIndex(prevStage.questions.length - 1);
    }
  };

  return (
    <div className="interview-engine">
      {/* 阶段进度条 */}
      <div className="stage-progress">
        {stages.map((stage, index) => (
          <div
            key={index}
            className={`stage-item ${
              index === currentStageIndex ? 'active' : ''
            } ${
              index < currentStageIndex ? 'completed' : ''
            }`}
          >
            阶段{index + 1}: {stage.name}
            {stage.questions.length > 0 && (
              <span className="question-count">
                ({currentQuestionIndex + 1}/{stage.questions.length})
              </span>
            )}
          </div>
        ))}
      </div>

      {/* 当前题目卡片 */}
      {currentStage.questions.length > 0 ? (
        <div className="question-card">
          <h2>阶段{currentStageIndex + 1}: {currentStage.name}</h2>
          <QuestionCard
            questionId={currentQuestionId}
            answer={userAnswers[currentQuestionId]}
            onAnswerChange={onAnswerChange}
          />

          {/* 过渡脚本显示 */}
          {currentQuestionIndex > 0 && getTransition() && (
            <div className="transition-script">
              <p>💭 {getTransition().script}</p>
              <small>连接：{getTransition().connection}</small>
            </div>
          )}

          {/* 按钮区域 */}
          <div className="button-group">
            <button
              onClick={handlePreviousQuestion}
              disabled={currentStageIndex === 0 && currentQuestionIndex === 0}
            >
              ← 上一题
            </button>

            {!isLastQuestionInStage ? (
              <button onClick={handleNextQuestion}>
                下一题 →
              </button>
            ) : !isLastStage ? (
              <button onClick={handleNextQuestion}>
                下一阶段 →
              </button>
            ) : (
              <button onClick={handleSubmit}>
                提交
              </button>
            )}
          </div>
        </div>
      ) : (
        <div className="stage-transition">
          <h2>阶段{currentStageIndex + 1}: {currentStage.name}</h2>
          <p>{currentStage.description || "过渡阶段"}</p>
          <button onClick={handleNextQuestion}>
            继续 →
          </button>
        </div>
      )}
    </div>
  );
};

export default InterviewEngine;
```

```vue
<!--

<template>
  <div class="interview-engine">
    <!-- 阶段进度条 -->
    <div class="stage-progress">
      <div
        v-for="(stage, index) in stages"
        :key="index"
        :class="[
          'stage-item',
          { active: index === currentStageIndex },
          { completed: index < currentStageIndex }
        ]"
      >
        阶段{{ index + 1 }}: {{ stage.name }}
        <span v-if="stage.questions.length > 0" class="question-count">
          ({{ currentQuestionIndex + 1 }}/{{ stage.questions.length }})
        </span>
      </div>
    </div>

    <!-- 当前题目卡片 -->
    <div v-if="currentStage.questions.length > 0" class="question-card">
      <h2>阶段{{ currentStageIndex + 1 }}: {{ currentStage.name }}</h2>

      <QuestionCard
        :questionId="currentQuestionId"
        :answer="userAnswers[currentQuestionId]"
        @answerChange="handleAnswerChange"
      />

      <!-- 过渡脚本显示 -->
      <div v-if="currentQuestionIndex > 0 && currentTransition" class="transition-script">
        <p>💭 {{ currentTransition.script }}</p>
        <small>连接：{{ currentTransition.connection }}</small>
      </div>

      <!-- 按钮区域 -->
      <div class="button-group">
        <button
          @click="handlePreviousQuestion"
          :disabled="currentStageIndex === 0 && currentQuestionIndex === 0"
        >
          ← 上一题
        </button>

        <button v-if="!isLastQuestionInStage" @click="handleNextQuestion">
          下一题 →
        </button>

        <button
          v-else-if="!isLastStage"
          @click="handleNextStage"
        >
          下一阶段 →
        </button>

        <button v-else @click="handleSubmit">
          提交
        </button>
      </div>
    </div>

    <!-- 过渡阶段 -->
    <div v-else class="stage-transition">
      <h2>阶段{{ currentStageIndex + 1 }}: {{ currentStage.name }}</h2>
      <p>{{ currentStage.description || '过渡阶段' }}</p>
      <button @click="handleNextStage">继续 →</button>
    </div>
  </div>
</template>

<script>
export default {
  name: 'InterviewEngine',

  props: {
    stages: {
      type: Array,
      required: true
    },
    userAnswers: {
      type: Object,
      required: true
    }
  },

  data() {
    return {
      currentStageIndex: 0,
      currentQuestionIndex: 0
    };
  },

  computed: {
    currentStage() {
      return this.stages[this.currentStageIndex];
    },

    currentQuestionId() {
      return this.currentStage.questions[this.currentQuestionIndex];
    },

    isLastQuestionInStage() {
      return this.currentQuestionIndex === this.currentStage.questions.length - 1;
    },

    isLastStage() {
      return this.currentStageIndex === this.stages.length - 1;
    },

    currentTransition() {
      if (this.isLastQuestionInStage) return null;

      const fromQuestion = this.currentStage.questions[this.currentQuestionIndex];
      const toQuestion = this.currentStage.questions[this.currentQuestionIndex + 1];
      const transitionKey = `${fromQuestion}_to_${toQuestion}`;

      return this.currentStage.transitions[transitionKey] || null;
    }
  },

  methods: {
    handleNextQuestion() {
      if (this.currentTransition) {
        console.log(`过渡：${this.currentTransition.script}`);
      }

      if (!this.isLastQuestionInStage) {
        this.currentQuestionIndex++;
      } else {
        this.handleNextStage();
      }
    },

    handlePreviousQuestion() {
      if (this.currentQuestionIndex > 0) {
        this.currentQuestionIndex--;
      } else if (this.currentStageIndex > 0) {
        this.currentStageIndex--;
        const prevStage = this.stages[this.currentStageIndex];
        this.currentQuestionIndex = prevStage.questions.length - 1;
      }
    },

    handleNextStage() {
      this.currentStageIndex++;
      this.currentQuestionIndex = 0;
    },

    handleAnswerChange(questionId, answer) {
      this.$emit('answerChange', { questionId, answer });
    },

    handleSubmit() {
      this.$emit('submit');
    }
  }
};
</script>
```
